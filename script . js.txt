let cart = [];
let currentFilter = 'all';
let currentSearch = '';

function renderProducts() {
  const grid = document.getElementById('productsGrid');
  const filtered = products.filter(p => {
    const matchesCat = currentFilter === 'all' || p.category === currentFilter;
    const q = currentSearch.toLowerCase().trim();
    const matchesSearch = !q || p.name.toLowerCase().includes(q) || p.desc.toLowerCase().includes(q);
    return matchesCat && matchesSearch;
  });

  if (filtered.length === 0) {
    grid.innerHTML = `<div style="grid-column:1/-1; text-align:center; padding:3rem; color:#6b7280;">
      لا توجد منتجات مطابقة لبحثك حالياً. انتظروا جديد منه وفاطمة قريباً! ✨
    </div>`;
    return;
  }

  grid.innerHTML = filtered.map(p => `
    <div class="product-card">
      <div class="card-banner">${p.emoji}</div>
      <div class="card-body">
        <h3>${p.name}</h3>
        <p class="card-desc">${p.desc}</p>
        <div class="product-price">${p.price} جنية</div>
        <div class="card-footer">
          <button class="btn-add-cart" onclick="addToCart('${p.id}')">+ أضف للسلة</button>
          <button class="btn-details" onclick="showSpecs('${p.id}')">التفاصيل</button>
        </div>
      </div>
    </div>
  `).join('');
}

// الفلاتر والبحث
document.getElementById('filterTabs').addEventListener('click', (e) => {
  if (!e.target.classList.contains('filter-btn')) return;
  document.querySelectorAll('.filter-btn').forEach(btn => btn.classList.remove('active'));
  e.target.classList.add('active');
  currentFilter = e.target.dataset.filter;
  renderProducts();
});

document.getElementById('searchInput').addEventListener('input', (e) => {
  currentSearch = e.target.value;
  renderProducts();
});

// سلة الطلبات
function addToCart(productId) {
  const prod = products.find(p => p.id === productId);
  if (!prod) return;

  const existing = cart.find(item => item.id === productId);
  if (existing) {
    existing.qty += 1;
  } else {
    cart.push({ ...prod, qty: 1 });
  }
  updateCartUI();
  showToast(`تمت إضافة "${prod.name}" إلى السلة`);
}

function changeQty(productId, delta) {
  const idx = cart.findIndex(i => i.id === productId);
  if (idx === -1) return;
  cart[idx].qty += delta;
  if (cart[idx].qty <= 0) {
    cart.splice(idx, 1);
  }
  updateCartUI();
}

function updateCartUI() {
  const totalItems = cart.reduce((sum, item) => sum + item.qty, 0);
  document.getElementById('cartCount').textContent = totalItems;

  const container = document.getElementById('cartItemsContainer');
  if (cart.length === 0) {
    container.innerHTML = `<div style="text-align:center; padding:2rem; color:#6b7280;">
      سلة الطلبات فارغة حالياً. اختاري ما يعجبك من المنتجات! 🛍️
    </div>`;
    return;
  }

  let totalSum = 0;
  container.innerHTML = cart.map(item => {
    totalSum += item.price * item.qty;
    return `
      <div class="cart-item">
        <div>
          <strong style="display:block; font-size:0.92rem;">${item.name}</strong>
          <small style="color:#ec4899; font-weight:700;">${item.price} ج × ${item.qty} = ${item.price * item.qty} ج</small>
        </div>
        <div class="qty-controls">
          <button class="qty-btn" onclick="changeQty('${item.id}', -1)">-</button>
          <strong>${item.qty}</strong>
          <button class="qty-btn" onclick="changeQty('${item.id}', 1)">+</button>
        </div>
      </div>
    `;
  }).join('') + `<div style="text-align:center; font-weight:900; margin-top:10px; color:#1f2937;">الإجمالي الكلي: ${totalSum} جنية</div>`;
}

function toggleDrawer(open) {
  const overlay = document.getElementById('drawerOverlay');
  if (open) overlay.classList.add('open');
  else overlay.classList.remove('open');
}

function buildOrderMessage() {
  const name = document.getElementById('clientName').value.trim() || "غير محدد";
  const phone = document.getElementById('clientPhone').value.trim() || "غير محدد";
  const address = document.getElementById('clientAddress').value.trim() || "غير محدد";

  let lines = [
    `🌸 *أوردر جديد من متجر FLORIDA* 🌸`,
    `اسم العميل: ${name}`,
    `رقم الهاتف: ${phone}`,
    `العنوان: ${address}`,
    `--------------------------`,
    `*المنتجات المطلوبة:*`
  ];

  let total = 0;
  if (cart.length === 0) {
    lines.push(`- لا توجد منتجات محددة`);
  } else {
    cart.forEach((item, i) => {
      lines.push(`${i + 1}. ${item.name} — الكمية: ${item.qty} — السعر: ${item.price * item.qty}ج`);
      total += item.price * item.qty;
    });
    lines.push(`--------------------------`);
    lines.push(`*إجمالي المبلغ المطلوب:* ${total} جنية`);
  }

  return lines.join('\n');
}

function submitOrder(e) {
  e.preventDefault();
  const msg = encodeURIComponent(buildOrderMessage());
  // استبدل الرقم أدناه برقم الواتساب الخاص بمنة أو فاطمة لاستقبال الطلبات مباشرة
  window.open(`https://wa.me/201000000000?text=${msg}`, '_blank');
}

function copyOrderText() {
  const text = buildOrderMessage();
  navigator.clipboard.writeText(text).then(() => {
    showToast("تم نسخ تفاصيل الأوردر بنجاح!");
  });
}

function showSpecs(productId) {
  const p = products.find(item => item.id === productId);
  if (!p) return;

  document.getElementById('modalContent').innerHTML = `

      <span style="font-size:2.5rem;">${p.emoji}</span>
      <button class="close-btn" onclick="closeModal()">×</button>
    </div>
    <h3 style="margin-bottom:0.5rem; color:#1f2937;">${p.name}</h3>
    <p style="color:#6b7280; font-size:0.92rem; margin-bottom:1rem;">${p.desc}</p>
    <p style="color:#ec4899; font-weight:900; font-size:1.2rem; margin-bottom:1rem;">السعر: ${p.price} جنية</p>
    <div style="background:#fdf2f8; padding:1rem; border-radius:10px; margin-bottom:1.2rem; font-size:0.9rem; color:#374151;">
      <strong>تفاصيل المنتج:</strong> ${p.details}
    </div>
    <button class="btn-primary" style="width:100%; justify-content:center;" onclick="addToCart('${p.id}'); closeModal();">
      + أضف المنتج للسلة
    </button>
  `;
  document.getElementById('specsModal').classList.add('open');
}

function closeModal() {
  document.getElementById('specsModal').classList.remove('open');
}

function showToast(msg) {
  const toast = document.getElementById('toast');
  toast.textContent = msg;
  toast.classList.add('show');
  setTimeout(() => toast.classList.remove('show'), 2800);
}
